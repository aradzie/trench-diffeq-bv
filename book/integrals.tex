\chapter{A Brief Table of Integrals}
\hypertarget{integrals}{}

\begin{description}
  \item $\dst\int u^{\alpha}\;du=\frac{u^{\alpha+1}}{\alpha+1}+c$, \quad $\alpha\ne-1$

  \item $\dst\int\frac{du}{u}=\ln|u|+c$

  \item $\dst\int\cos u\;du=\sin u+c$

  \item $\dst\int\sin u\;du=-\cos u+c$

  \item $\dst\int\tan u\;du=-\ln|\cos u|+c$

  \item $\dst\int\cot u\;du=\ln|\sin u|+c$

  \item $\dst\int\sec^{2}u\;du=\tan u+c$

  \item $\dst\int\csc^{2}u\;du=-\cot u+c$

  \item $\dst\int\sec u\;du=\ln|\sec u+\tan u|+c$

  \item $\dst\int\cos^{2}u\;du=\frac{u}{2}+\frac{1}{4}\sin2u+c$

  \item $\dst\int\sin^{2}u\;du=\frac{u}{2}-\frac{1}{4}\sin2u+c$

  \item $\dst\int\frac{du}{1+u^{2}}\;du=\tan^{-1}u+c$

  \item $\dst\int\frac{du}{\sqrt{1-u^{2}}}\;du=\sin^{-1}u+c$

  \item $\dst\int\frac{1}{u^{2}-1}\;du=\frac{1}{2}\ln\left|\frac{u-1}{u+1}\right|+c$

  \item $\dst\int\cosh u\;du=\sinh u+c$

  \item $\dst\int\sinh u\;du=\cosh u+c$

  \item $\dst\int u\;dv=uv-\int v\;du$

  \item $\dst\int u\cos u\;du=u\sin u+\cos u+c$

  \item $\dst\int u\sin u\;du=-u\cos u+\sin u+c$

  \item $\dst\int ue^{u}\;du=ue^{u}-e^{u}+c$

  \item $\dst\int e^{\lambda u}\cos\omega u\;du=\frac{e^{\lambda u}(\lambda\cos\omega u+\omega\sin\omega u)}{\lambda^{2}+\omega^{2}}+c$

  \item $\dst\int e^{\lambda u}\sin\omega u\;du=\frac{e^{\lambda u}(\lambda\sin\omega u-\omega\cos\omega u)}{\lambda^{2}+\omega^{2}}+c$

  \item $\dst\int\ln|u|\;du=u\ln|u|-u+c$

  \item $\dst\int u\ln|u|\;du=\frac{u^{2}\ln|u|}{2}-\frac{u^{2}}{4}+c$

  \item $\dst\int\cos\omega_{1}u\cos\omega_{2}u\,du=\frac{\sin(\omega_{1}+\omega_{2})u}{2(\omega_{1}+\omega_{2})}+\frac{\sin(\omega_{1}-\omega_{2})u}{2(\omega_{1}-\omega_{2})}+c\quad(\omega_{1}\ne\pm\omega_{2})$

  \item $\dst\int\sin\omega_{1}u\sin\omega_{2}u\,du=-\frac{\sin(\omega_{1}+\omega_{2})u}{2(\omega_{1}+\omega_{2})}+\frac{\sin(\omega_{1}-\omega_{2})u}{2(\omega_{1}-\omega_{2})}+c\quad(\omega_{1}\ne\pm\omega_{2})$

  \item $\dst\int\sin\omega_{1}u\cos\omega_{2}u\,du=-\frac{\cos(\omega_{1}+\omega_{2})u}{2(\omega_{1}+\omega_{2})}-\frac{\cos(\omega_{1}-\omega_{2})u}{2(\omega_{1}-\omega_{2})}+c\quad(\omega_{1}\ne\pm\omega_{2})$
\end{description}
