\answer{6.4.1}

 $Y^{2}=\rho(\rho-2X)$;   if $e\ne1$
$\dst{\left(X+\frac{e\rho}{1-e^{2}}\right)^{2}}+\dst\frac{Y^{2}}{1-e^{2}}=
\dst\frac{\rho^{2}}{(1-e^{2})^{2}}$   if
\vspace*{-.07in}; \begin{quote}
$e<1$ let
$X_{0}=-\dst\frac{e\rho}{1-e^{2}}$,
$a=\dst\frac{\rho}{1-e^{2}}$,  $b=\dst\frac{\rho}{\sqrt{1-e^{2}}}$.
\end{quote}
