\vspace*{-.07in}; \begin{quote}
$e<1$ let
$X_0=-\dst{e\rho\over1-e^2}$,
$a=\dst{\rho\over1-e^2}$,  $b=\dst{\rho\over\sqrt{1-e^2}}$.
\end{quote}

\end{quote}
